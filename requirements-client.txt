# Minimal client-side deps for running scripts/qsql_query.py against an
# already-running MCP server over HTTP (default: http://localhost:9000/mcp).
# For self-hosting the actual server instead, see requirements-server.txt
# (and requirements-vllm.txt for the vLLM model host it talks to).
#
# This is the "just call the tool" path -- it does NOT self-host the MCP
# server, so it needs none of pykx / vllm / qlint / starlette-server bits.
# The `mcp` package pulls in httpx + anyio transitively.
#
# Install into a fresh env, e.g.:
#   conda create -n <your-client-env> python=3.12 -y
#   conda activate <your-client-env>
#   pip install -r requirements-client.txt
#
# NOTE: If you instead run qsql_query.py with --stdio (spawn a local server),
# you need the FULL server stack: requirements-server.txt PLUS a manual pykx
# install PLUS QLINT_DIR + a running vLLM.

# Pin to 1.x: mcp 2.0 renamed `streamablehttp_client` -> `streamable_http_client`,
# which breaks qsql_query.py's import (the script uses the 1.x name).
mcp>=1.27,<2
